# -> conv 10x10 -> pool 5x5
final_length = 5

if verbose:
    print(f"final_length = {final_length}")

class LeNet5(nn.Module):
    def __init__(self, num_classes, grayscale=False):
        super(LeNet5, self).__init__()
        self.grayscale = grayscale
        self.num_classes = num_classes
        in_channels = 1 if self.grayscale else 3

        self.features = nn.Sequential(
            nn.Conv2d(in_channels, layer_1_n_filters,
                      kernel_size=kernel_size, padding=2),
            nn.Tanh(),
            nn.MaxPool2d(kernel_size=2, stride=2),

            nn.Conv2d(layer_1_n_filters, layer_2_n_filters,
                      kernel_size=kernel_size, padding=0),
            nn.Tanh(),
            nn.MaxPool2d(kernel_size=2, stride=2),
        )

        self.classifier = nn.Sequential(
            nn.Linear(final_length * final_length * layer_2_n_filters,
                      fc_1_n_nodes),
            nn.Tanh(),
            nn.Linear(fc_1_n_nodes, fc_2_n_nodes),
            nn.Tanh(),
            nn.Linear(fc_2_n_nodes, num_classes),
        )

    def forward(self, x):
        x = self.features(x)
        x = x.view(x.size(0), -1)
        logits = self.classifier(x)
        probas = F.softmax(logits, dim=1)
        return logits, probas

def train(epoch):
    model.train()
    criterion = nn.CrossEntropyLoss()
    running_loss = 0.0
    correct = 0
    total = 0

    for batch_idx, (data, target) in enumerate(train_loader):
        if cuda:
            data, target = data.cuda(), target.cuda()

        optimizer.zero_grad()
        logits, probas = model(data)
        loss = criterion(logits, target)
        loss.backward()
        optimizer.step()

        running_loss += loss.item()
        pred = logits.argmax(dim=1)
        correct += (pred == target).sum().item()
        total += target.size(0)

        if batch_idx % logging_interval == 0:
            print(f'Train Epoch {epoch} '
                  f'[{batch_idx * len(data)}/{len(train_loader.dataset)}] '
                  f'Loss: {loss.item():.6f}')

    train_loss = running_loss / len(train_loader)
    train_acc = correct / total
    writer.add_scalar('Loss/train', train_loss, epoch)
    writer.add_scalar('Accuracy/train', train_acc, epoch)
    print(f"Epoch {epoch}: train_loss={train_loss:.4f}, train_acc={train_acc:.4f}")

    # log model parameters
    for name, param in model.named_parameters():
        writer.add_histogram(f'weights/{name}', param.data.cpu().numpy(), epoch)

def test(epoch):
    model.eval()
    criterion = nn.CrossEntropyLoss()
    test_loss = 0.0
    correct = 0

    with torch.no_grad():
        for data, target in test_loader:
            if cuda:
                data, target = data.cuda(), target.cuda()

            logits, probas = model(data)
            test_loss += criterion(logits, target).item()
            pred = logits.argmax(dim=1)
            correct += (pred == target).sum().item()

    test_loss /= len(test_loader)
    test_acc = correct / len(test_loader.dataset)
    writer.add_scalar('Loss/test', test_loss, epoch)
    writer.add_scalar('Accuracy/test', test_acc, epoch)
    print(f"Test epoch {epoch}: loss={test_loss:.4f}, "
          f"acc={100 * test_acc:.2f}%\n")

1(c) Filter and activation observations¶

The six 5×5 first-layer kernels show oriented edge / Gabor-like patterns: some respond to horizontal edges, others to vertical or diagonal edges, and some act as center-surround blob detectors. Different filters also exhibit different spatial frequencies and opposite polarity (light-on-dark versus dark-on-light). This is exactly the classical "bank of edge detectors" that is expected from the first layer of a CNN trained on natural images.

On the test set, conv1 activations have mean ≈ −0.014 and std ≈ 0.326, while conv2 activations have mean ≈ −0.210 and std ≈ 0.444. The histograms are unimodal and roughly centred near zero, with heavier tails in conv2, indicating that deeper features are sparser and have a larger dynamic range.

def make_cifar_loaders(bs):
    tr = DataLoader(train_dataset, batch_size=bs, shuffle=True, num_workers=0)
    te = DataLoader(test_dataset, batch_size=bs, shuffle=False, num_workers=0)
    return tr, te

def run_cifar_experiment(lr=0.001, bs=128, opt_name='Adam', n_epochs=3, tag=None):
    tr_loader, te_loader = make_cifar_loaders(bs)
    m = LeNet5(num_classes=10, grayscale=True)
    if cuda:
        m.cuda()

    if opt_name == 'Adam':
        opt = optim.Adam(m.parameters(), lr=lr)
    elif opt_name == 'SGD':
        opt = optim.SGD(m.parameters(), lr=lr, momentum=0.9)
    else:
        raise ValueError(opt_name)

    criterion = nn.CrossEntropyLoss()
    if tag is None:
        tag = f'{opt_name}_lr{lr}_bs{bs}'

    hist = {'train_loss': [], 'train_acc': [], 'test_loss': [], 'test_acc': []}

    for epoch in range(1, n_epochs + 1):
        m.train()
        running_loss, correct, total = 0.0, 0, 0
        for data, target in tr_loader:
            if cuda:
                data, target = data.cuda(), target.cuda()
            opt.zero_grad()
            logits, _ = m(data)
            loss = criterion(logits, target)
            loss.backward()
            opt.step()
            running_loss += loss.item()
            correct += (logits.argmax(1) == target).sum().item()
            total += target.size(0)
        train_loss = running_loss / len(tr_loader)
        train_acc = correct / total

        m.eval()
        te_loss, te_correct = 0.0, 0
        with torch.no_grad():
            for data, target in te_loader:
                if cuda:
                    data, target = data.cuda(), target.cuda()
                logits, _ = m(data)
                te_loss += criterion(logits, target).item()
                te_correct += (logits.argmax(1) == target).sum().item()
        te_loss /= len(te_loader)
        te_acc = te_correct / len(te_loader.dataset)

        hist['train_loss'].append(train_loss)
        hist['train_acc'].append(train_acc)
        hist['test_loss'].append(te_loss)
        hist['test_acc'].append(te_acc)

        writer.add_scalar(f'{tag}/Loss/train', train_loss, epoch)
        writer.add_scalar(f'{tag}/Accuracy/train', train_acc, epoch)
        writer.add_scalar(f'{tag}/Loss/test', te_loss, epoch)
        writer.add_scalar(f'{tag}/Accuracy/test', te_acc, epoch)
        print(f"{tag} epoch {epoch}: train_acc={train_acc:.4f}, test_acc={te_acc:.4f}")

    return hist

1(b) Hyperparameter search — observations¶

Four configurations were compared for 3 epochs each: Adam with lr = 1e-3, 1e-2, and 1e-4, and SGD with momentum 0.9 and lr = 1e-2. Adam with lr = 1e-3 achieved the best final test accuracy (43.7%), followed by Adam lr = 1e-2 (38.3%), SGD with momentum (32.5%) and Adam lr = 1e-4 (27.3%).

A learning rate that is too small (1e-4) converges too slowly within a short budget. Adam lr = 1e-2 reached a higher train accuracy by epoch 3 (45.4%) but a lower test accuracy (38.3%), suggesting less stable generalization in this short run. SGD with momentum needs more epochs or tuning to match Adam. The 10-epoch baseline at Adam lr = 1e-3 reached 49.6% test accuracy.

Visualizing and Understanding Convolutional Networks (Zeiler & Fergus, 2014)¶

2. Summary of "Visualizing and Understanding Convolutional Networks"¶

(Zeiler & Fergus, ECCV 2014)¶

The paper asks a simple but important question: what do convolutional networks actually learn, and can we use that understanding to build better models? The authors introduce a multi-faceted visualisation framework and use it both as a diagnostic and as a guide for architecture design.

(1) Deconvolutional feature visualisation. They attach a "deconvnet" to any convolutional layer. Given an activation map, the deconvnet performs unpooling (using the arg-max "switch" locations recorded during the forward max-pooling pass), rectification (ReLU), and transposed convolution with the same filters, mapping the activation back to input-pixel space. The resulting image shows the pattern that maximally excites a given unit. The visualisations reveal a clear hierarchy: layer 1–2 units detect edges, colours and simple textures; layer 3 captures more complex textures and object parts; layers 4–5 encode whole object parts and entire objects with pose variation. The analysis also exposes dead filters and aliasing artefacts caused by large strides.

(2) Occlusion sensitivity. A grey square is slid over the input image and the drop in the correct-class score is recorded, producing a heat map of which image regions the model relies on. The heat maps show that the network localises objects rather than relying on background context, and they can even be used for weakly-supervised localisation.

(3) Feature evolution during training. Running the deconvnet at different epochs shows that lower layers converge within a few epochs (generic edges), while higher layers need many epochs to become class-specific. This explains why fine-tuning only the top layers is often sufficient.

(4) Architecture insights. Motivated by the visualisations, they replace the 11×11 stride-4 first-layer filters with 7×7 stride-2 filters, increase the number of feature maps, add a sixth convolutional layer, and reduce stride elsewhere. These changes improved ImageNet 2012 top-5 error from 16.4% (Krizhevsky et al.) to 14.8%, and the resulting features transferred well to Caltech-256, Caltech-101 and PASCAL VOC, achieving state-of-the-art results at the time.

(5) Ablations. Clamping high-activation features, removing layers, or perturbing filters degrades accuracy in a way that correlates with the visualisations, confirming that the visualised features are genuinely used by the classifier.

Key takeaway. Visualisation is not merely cosmetic: deconvnet feature maps and occlusion heat maps provide actionable insight into what a CNN has learned, how features evolve during training, and where the architecture can be improved.

def occlusion_heatmap(model, img, label, patch=8, stride=4):
    model.eval()
    H, W = img.shape[-2:]
    heat = np.zeros((H, W)); counts = np.zeros((H, W))
    inp = img.unsqueeze(0)
    if cuda: inp = inp.cuda()
    with torch.no_grad():
        base = F.softmax(model(inp)[0], 1)[0, label].item()
        for i in range(0, H - patch + 1, stride):
            for j in range(0, W - patch + 1, stride):
                occ = img.clone()
                occ[:, i:i+patch, j:j+patch] = 0.0
                o = occ.unsqueeze(0)
                if cuda: o = o.cuda()
                p = F.softmax(model(o)[0], 1)[0, label].item()
                heat[i:i+patch, j:j+patch] += (base - p)
                counts[i:i+patch, j:j+patch] += 1
    return heat / np.maximum(counts, 1)
